module Ritka (
    input        clk,
    input        btn1,
    input        btn2,
    input        btn_sw,
    output       led1,
    output       led2,
    input        rx,
    output       tx
);

    // =========================================================
    // UART
    // =========================================================
    wire rxclk_en;
    wire txclk_en;

    reg        rdy_clr;
    wire       rdy;
    wire [7:0] dout;

    reg        wr_en;
    reg [7:0]  din;
    wire       tx_busy;

    baud_rate_gen uart_baud (
        .clk_50m (clk),
        .rxclk_en(rxclk_en),
        .txclk_en(txclk_en)
    );

    transmitter uart_tx (
        .din     (din),
        .wr_en   (wr_en),
        .clk_50m (clk),
        .clken   (txclk_en),
        .tx      (tx),
        .tx_busy (tx_busy)
    );

    receiver uart_rx (
        .rx      (rx),
        .rdy     (rdy),
        .rdy_clr (rdy_clr),
        .clk_50m (clk),
        .clken   (rxclk_en),
        .data    (dout)
    );

    // =========================================================
    // RDY posedge
    // =========================================================
    reg rdy_d;
    always @(posedge clk) rdy_d <= rdy;
    wire rdy_posedge = rdy & ~rdy_d;

    // Detect space character from RX
    reg space_pulse;
    always @(posedge clk) begin
        space_pulse <= 1'b0;
        if (rdy_posedge && dout == 8'h20)
            space_pulse <= 1'b1;
    end

    always @(posedge clk)
        rdy_clr <= rdy_posedge;

    // =========================================================
    // Button sync + edge detection
    // =========================================================
    reg btn1_d1, btn1_d2;
    reg btn2_d1, btn2_d2;
    reg btn_sw_d1, btn_sw_d2;

    always @(posedge clk) begin
        btn1_d1   <= btn1;    btn1_d2   <= btn1_d1;
        btn2_d1   <= btn2;    btn2_d2   <= btn2_d1;
        btn_sw_d1 <= btn_sw;  btn_sw_d2 <= btn_sw_d1;
    end

    wire btn1_pos   = btn1_d1   & ~btn1_d2;
    wire btn2_pos   = btn2_d1   & ~btn2_d2;
    wire btn_sw_pos = btn_sw_d1 & ~btn_sw_d2;

    // =========================================================
    // Button state register (captures button states)
    // =========================================================
    reg [2:0] button_state;  // {btn1, btn2, btn_sw} - старший, средний, младший
    reg [2:0] saved_state;   // Сохраненное состояние для отправки
    
    always @(posedge clk) begin
        button_state <= {btn1_d1, btn2_d1, btn_sw_d1};
        
        // Save button state when space is pressed
        if (space_pulse) begin
            saved_state <= button_state;
        end
    end

    // =========================================================
    // LED registers (for visual feedback)
    // =========================================================
    reg led1_r, led2_r;
    
    always @(posedge clk) begin
        // LEDs show current button state
        led1_r <= button_state[2];  // Старший бит
        led2_r <= button_state[1];  // Средний бит
    end

    // =========================================================
    // TX FSM for Roman numerals
    // =========================================================
    reg [2:0] tx_state;
    reg [2:0] tx_idx;
    reg sending;

    localparam TX_IDLE  = 3'd0,
               TX_ROMAN = 3'd1;

    always @(posedge clk) begin
        wr_en <= 1'b0;

        // Start transmission when space is pressed
        if (space_pulse) begin
            tx_state <= TX_ROMAN;
            tx_idx   <= 0;
            sending  <= 0;
        end

        // ----------- UART TX FSM -----------
        if (tx_state != TX_IDLE) begin
            if (!tx_busy && !sending) begin
                case (tx_state)
                    TX_ROMAN: begin
                        case (saved_state)
                            // Decimal 0 (000)
                            3'b000: begin
                                if      (tx_idx == 0) din <= 8'h4E; // N
                                else if (tx_idx == 1) din <= 8'h55; // U
                                else if (tx_idx == 2) din <= 8'h4C; // L
                                else if (tx_idx == 3) din <= 8'h4C; // L
                                else if (tx_idx == 4) din <= 8'h41; // A
                            end
                            // Decimal 1 (001)
                            3'b001: begin
                                if      (tx_idx == 0) din <= 8'h49; // I
                            end
                            // Decimal 2 (010)
                            3'b010: begin
                                if      (tx_idx == 0) din <= 8'h49; // I
                                else if (tx_idx == 1) din <= 8'h49; // I
                            end
                            // Decimal 3 (011)
                            3'b011: begin
                                if      (tx_idx == 0) din <= 8'h49; // I
                                else if (tx_idx == 1) din <= 8'h49; // I
                                else if (tx_idx == 2) din <= 8'h49; // I
                            end
                            // Decimal 4 (100)
                            3'b100: begin
                                if      (tx_idx == 0) din <= 8'h49; // I
                                else if (tx_idx == 1) din <= 8'h56; // V
                            end
                            // Decimal 5 (101)
                            3'b101: begin
                                if      (tx_idx == 0) din <= 8'h56; // V
                            end
                            // Decimal 6 (110)
                            3'b110: begin
                                if      (tx_idx == 0) din <= 8'h56; // V
                                else if (tx_idx == 1) din <= 8'h49; // I
                            end
                            // Decimal 7 (111)
                            3'b111: begin
                                if      (tx_idx == 0) din <= 8'h56; // V
                                else if (tx_idx == 1) din <= 8'h49; // I
                                else if (tx_idx == 2) din <= 8'h49; // I
                            end
                        endcase

                        wr_en   <= 1'b1;
                        sending <= 1'b1;
                    end
                endcase
            end

            if (sending && tx_busy) begin
                sending <= 0;
                tx_idx <= tx_idx + 1;

                // Check if we've sent all characters for this number
                case (saved_state)
                    3'b000: begin // NULLA (5 chars)
                        if (tx_idx == 4) tx_state <= TX_IDLE;
                    end
                    3'b001: begin // I (1 char)
                        if (tx_idx == 0) tx_state <= TX_IDLE;
                    end
                    3'b010: begin // II (2 chars)
                        if (tx_idx == 1) tx_state <= TX_IDLE;
                    end
                    3'b011: begin // III (3 chars)
                        if (tx_idx == 2) tx_state <= TX_IDLE;
                    end
                    3'b100: begin // IV (2 chars)
                        if (tx_idx == 1) tx_state <= TX_IDLE;
                    end
                    3'b101: begin // V (1 char)
                        if (tx_idx == 0) tx_state <= TX_IDLE;
                    end
                    3'b110: begin // VI (2 chars)
                        if (tx_idx == 1) tx_state <= TX_IDLE;
                    end
                    3'b111: begin // VII (3 chars)
                        if (tx_idx == 2) tx_state <= TX_IDLE;
                    end
                endcase
            end
        end
    end

    assign led1 = led1_r;
    assign led2 = led2_r;

endmodule
