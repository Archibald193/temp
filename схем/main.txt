module lab3_1(
    input wire clk,
    input wire rst,
    output reg test_led9,
    output wire tx,
    input wire rx,
    input wire pin_88,//или pin_144
    input wire pin_24,
    input wire pin_25
);

reg [31:0] cnt;
wire clr_cnt;
wire txclk_en;
wire rxclk_en;
wire [7:0] rx_data;
wire rx_rdy;

assign clr_cnt = (cnt == 32'd25_000_000);

always @(posedge clk or negedge rst) begin
    if(!rst) cnt <= 32'h0;
    else if(clr_cnt) cnt <= 32'h0;
    else cnt <= cnt + 32'h1;
end

always @(posedge clk or negedge rst) begin
    if(!rst) test_led9 <= 1'b0;
    else if (clr_cnt) test_led9 <= ~test_led9;
end

baud_rate_gen uart_baud(.clk_50m(clk), .rxclk_en(rxclk_en), .txclk_en(txclk_en));
transmitter uart_tx(.din(tx_byte), .wr_en(tx_start), .clk_50m(clk), .clken(txclk_en), .tx(tx));
receiver uart_rx(.rx(rx), .rdy(rx_rdy), .rdy_clr(rx_rdy), .clk_50m(clk), .clken(rxclk_en), .data(rx_data));

reg [7:0] tx_byte;
reg tx_start;
reg [2:0] button_val;
reg [3:0] char_index;
reg [2:0] state;
reg [15:0] delay_timer;

localparam IDLE      = 0;
localparam GET_CHAR  = 1; 
localparam SEND_CHAR = 2;
localparam WAIT_TX   = 3;
wire [2:0] current_buttons = {~pin_88, ~pin_24, ~pin_25};
always @(posedge clk or negedge rst) begin
    if (!rst) begin
        state <= IDLE;
        tx_start <= 0;
        char_index <= 0;
        button_val <= 0;
        tx_byte <= 0;
    end else begin
        case (state)
            IDLE: begin
                tx_start <= 0;
                char_index <= 0;
                if (rx_rdy && rx_data == 8'h20) begin
                    button_val <= current_buttons;
                    state <= GET_CHAR;
                end
            end

            GET_CHAR: begin
                case (button_val)
                    3'd0: begin
                        if (char_index == 0) tx_byte <= "0"; 
                        else state <= IDLE; 
                    end
                    3'd1: begin
                        if (char_index == 0) tx_byte <= "I"; 
                        else state <= IDLE; 
                    end
                    3'd2: begin
                        if (char_index < 2) tx_byte <= "I"; 
                        else state <= IDLE; 
                    end
                    3'd3: begin
                        if (char_index < 3) tx_byte <= "I"; 
                        else state <= IDLE; 
                    end
                    3'd4: begin
                        if (char_index == 0) tx_byte <= "I";
                        else if (char_index == 1) tx_byte <= "V";
                        else state <= IDLE; 
                    end
                    3'd5: begin
                        if (char_index == 0) tx_byte <= "V"; 
                        else state <= IDLE; 
                    end
                    3'd6: begin
                        if (char_index == 0) tx_byte <= "V";
                        else if (char_index == 1) tx_byte <= "I";
                        else state <= IDLE; 
                    end
                    3'd7: begin
                        if (char_index == 0) tx_byte <= "V";
                        else if (char_index < 3) tx_byte <= "I";
                        else state <= IDLE; 
                    end
                endcase

                if (state == GET_CHAR) state <= SEND_CHAR;
            end

            SEND_CHAR: begin
                tx_start <= 1;
                state <= WAIT_TX;
                delay_timer <= 0;
            end
            WAIT_TX: begin
                tx_start <= 0;
                if (delay_timer < 16'd10000) begin
                    delay_timer <= delay_timer + 1'b1;
                end else begin
                    char_index <= char_index + 1'b1;
                    state <= GET_CHAR;
                end
            end
        endcase
    end
end

endmodule
