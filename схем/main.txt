#include "main.h"
#include "init.h"
#include <stdio.h>

// --- Глобальные переменные ---
volatile uint32_t ms_ticks = 0;

// Статусы светодиодов: 1 - мигает, 0 - выключен
volatile uint8_t led_active[4] = {0, 0, 0, 0};

// Настройка по заданию:
// 1 - Оранжевый (PD13)
// 2 - Красный (PD14)
// 3 - Синий (PD15)
// 4 - Зеленый (PD12)
uint16_t LED_PINS[4] = {GPIO_Pin_13, GPIO_Pin_14, GPIO_Pin_15, GPIO_Pin_12};
char* LED_NAMES[4] = {"ORANGE", "RED",   "BLUE",   "GREEN"};

// Для printf
int fputc(int ch, FILE *f) {
    USART_SendData(USART2, (uint8_t) ch);
    while (USART_GetFlagStatus(USART2, USART_FLAG_TC) == RESET) {}
    return ch;
}

void SysTick_Handler(void) {
    ms_ticks++;
}

int main(void) {
    // Объявление переменных В НАЧАЛЕ функции (чтобы не было error #29)
    uint32_t last_blink_time = 0;
    uint8_t blink_state = 0;
    int i; 
    char received_char;
    int led_index;

    SystemInit(); 
    SystemCoreClockUpdate(); 

    LEDs_ini();
    USART2_ini(); 
    
    SysTick_Config(SystemCoreClock / 1000); 

    printf("\r\n--- WORKING MODE (POLLING) ---\r\n");
    printf("1: ORANGE\r\n2: RED\r\n3: BLUE\r\n4: GREEN\r\n");

    while(1) {
        
        // --- ЧАСТЬ 1: ПРОВЕРКА КЛАВИАТУРЫ (POLLING) ---
        // Проверяем флаг RXNE (Receive Not Empty). Если он = 1, значит пришел символ.
        if (USART_GetFlagStatus(USART2, USART_FLAG_RXNE) != RESET) {
            
            received_char = (char)USART_ReceiveData(USART2);
            led_index = -1;

            switch (received_char) {
                case '1': led_index = 0; break; // Оранжевый
                case '2': led_index = 1; break; // Красный
                case '3': led_index = 2; break; // Синий
                case '4': led_index = 3; break; // Зеленый
                default: break; 
            }

            if (led_index != -1) {
                // Переключаем режим (Вкл / Выкл)
                led_active[led_index] = !led_active[led_index];

                printf("\r\n[%s] Status: ", LED_NAMES[led_index]);
                if (led_active[led_index]) {
                    printf("BLINKING");
                } else {
                    printf("STOPPED");
                    // Сразу гасим физически
                    GPIO_ResetBits(GPIOD, LED_PINS[led_index]);
                }
            }
        }

        // --- ЧАСТЬ 2: МИГАНИЕ СВЕТОДИОДАМИ ---
        if ((ms_ticks - last_blink_time) >= 200) {
            last_blink_time = ms_ticks;
            blink_state = !blink_state; // 0 -> 1 -> 0

            // Пробегаем по всем светодиодам
            for (i = 0; i < 4; i++) {
                if (led_active[i]) {
                    // Если режим "Активен" - мигаем
                    if (blink_state) {
                        GPIO_SetBits(GPIOD, LED_PINS[i]);
                    } else {
                        GPIO_ResetBits(GPIOD, LED_PINS[i]);
                    }
                } else {
                    // Если режим "Стоп" - держим выключенным
                    GPIO_ResetBits(GPIOD, LED_PINS[i]);
                }
            }
        }
    }
}
