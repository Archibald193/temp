// --- Инициализация светодиодов (GPIOD 12, 13, 14, 15) ---
void LEDs_ini(void) {
    GPIO_InitTypeDef GPIO_Init_LED;
    RCC_AHB1PeriphClockCmd(RCC_AHB1Periph_GPIOD, ENABLE);

    GPIO_Init_LED.GPIO_Pin = GPIO_Pin_12 | GPIO_Pin_13 | GPIO_Pin_14 | GPIO_Pin_15;
    GPIO_Init_LED.GPIO_Mode = GPIO_Mode_OUT;
    GPIO_Init_LED.GPIO_Speed = GPIO_Speed_2MHz;
    GPIO_Init_LED.GPIO_OType = GPIO_OType_PP;
    GPIO_Init_LED.GPIO_PuPd = GPIO_PuPd_NOPULL;
    GPIO_Init(GPIOD, &GPIO_Init_LED);
}

// --- Инициализация UART2 (PA2 = TX, PA3 = RX) ---
void USART2_ini(void) {
    GPIO_InitTypeDef GPIO_InitStruct;
    USART_InitTypeDef USART_InitStruct;

    // 1. Включаем тактирование
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_USART2, ENABLE); // UART2 на шине APB1
    RCC_AHB1PeriphClockCmd(RCC_AHB1Periph_GPIOA, ENABLE);  // Порт А

    // 2. Настраиваем Пины PA2 и PA3 как Альтернативную функцию (AF)
    GPIO_InitStruct.GPIO_Pin = GPIO_Pin_2 | GPIO_Pin_3;
    GPIO_InitStruct.GPIO_Mode = GPIO_Mode_AF;      // Важно! Режим AF
    GPIO_InitStruct.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_InitStruct.GPIO_OType = GPIO_OType_PP;
    GPIO_InitStruct.GPIO_PuPd = GPIO_PuPd_UP;
    GPIO_Init(GPIOA, &GPIO_InitStruct);

    // 3. Соединяем пины с модулем UART внутри чипа
    GPIO_PinAFConfig(GPIOA, GPIO_PinSource2, GPIO_AF_USART2); // PA2 -> TX
    GPIO_PinAFConfig(GPIOA, GPIO_PinSource3, GPIO_AF_USART2); // PA3 -> RX

    // 4. Настройки UART
    USART_InitStruct.USART_BaudRate = 9600; // Скорость 9600
    USART_InitStruct.USART_WordLength = USART_WordLength_8b;
    USART_InitStruct.USART_StopBits = USART_StopBits_1;
    USART_InitStruct.USART_Parity = USART_Parity_No;
    USART_InitStruct.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_InitStruct.USART_Mode = USART_Mode_Rx | USART_Mode_Tx;
    
    USART_Init(USART2, &USART_InitStruct);
    USART_Cmd(USART2, ENABLE); // Включаем UART
}

// Функция отправки строки (вместо printf, чтобы не зависало)
void USART_Puts(char *str) {
    while (*str) {
        while (USART_GetFlagStatus(USART2, USART_FLAG_TC) == RESET);
        USART_SendData(USART2, *str++);
    }
}

int main(void) {
    SystemInit();

    LEDs_ini();
    USART2_ini();

    // Приветствие (если это придет в терминал - значит связь есть)
    USART_Puts("System Started. Press 1-4 to toggle LEDs.\r\n");

    while(1) {
        // Проверяем, пришел ли байт
        if (USART_GetFlagStatus(USART2, USART_FLAG_RXNE) != RESET) {
            
            // Читаем байт
            char c = (char)USART_ReceiveData(USART2);
            
            // Эхо (отправляем обратно, чтобы видеть, что нажали)
            while (USART_GetFlagStatus(USART2, USART_FLAG_TC) == RESET);
            USART_SendData(USART2, c);

            // Управление
            switch (c) {
                case '1': 
                    GPIOD->ODR ^= GPIO_Pin_12; // Инверсия (вкл/выкл) Зеленый
                    USART_Puts(" -> Green Toggled\r\n");
                    break;
                case '2': 
                    GPIOD->ODR ^= GPIO_Pin_13; // Оранжевый
                    USART_Puts(" -> Orange Toggled\r\n");
                    break;
                case '3': 
                    GPIOD->ODR ^= GPIO_Pin_14; // Красный
                    USART_Puts(" -> Red Toggled\r\n");
                    break;
                case '4': 
                    GPIOD->ODR ^= GPIO_Pin_15; // Синий
                    USART_Puts(" -> Blue Toggled\r\n");
                    break;
            }
        }
    }

reg [7:0] tx_byte;
reg tx_start;
reg [2:0] button_val;
reg [3:0] char_index;
reg [2:0] state;
reg [15:0] delay_timer;

localparam IDLE      = 0;
localparam GET_CHAR  = 1; 
localparam SEND_CHAR = 2;
localparam WAIT_TX   = 3;
wire [2:0] current_buttons = {~pin_88, ~pin_24, ~pin_25};
always @(posedge clk or negedge rst) begin
    if (!rst) begin
        state <= IDLE;
        tx_start <= 0;
        char_index <= 0;
        button_val <= 0;
        tx_byte <= 0;
    end else begin
        case (state)
            IDLE: begin
                tx_start <= 0;
                char_index <= 0;
                if (rx_rdy && rx_data == 8'h20) begin
                    button_val <= current_buttons;
                    state <= GET_CHAR;
                end
            end

            GET_CHAR: begin
                case (button_val)
                    3'd0: begin
                        if (char_index == 0) tx_byte <= "0"; 
                        else state <= IDLE; 
                    end
                    3'd1: begin
                        if (char_index == 0) tx_byte <= "I"; 
                        else state <= IDLE; 
                    end
                    3'd2: begin
                        if (char_index < 2) tx_byte <= "I"; 
                        else state <= IDLE; 
                    end
                    3'd3: begin
                        if (char_index < 3) tx_byte <= "I"; 
                        else state <= IDLE; 
                    end
                    3'd4: begin
                        if (char_index == 0) tx_byte <= "I";
                        else if (char_index == 1) tx_byte <= "V";
                        else state <= IDLE; 
                    end
                    3'd5: begin
                        if (char_index == 0) tx_byte <= "V"; 
                        else state <= IDLE; 
                    end
                    3'd6: begin
                        if (char_index == 0) tx_byte <= "V";
                        else if (char_index == 1) tx_byte <= "I";
                        else state <= IDLE; 
                    end
                    3'd7: begin
                        if (char_index == 0) tx_byte <= "V";
                        else if (char_index < 3) tx_byte <= "I";
                        else state <= IDLE; 
                    end
                endcase

                if (state == GET_CHAR) state <= SEND_CHAR;
            end

            SEND_CHAR: begin
                tx_start <= 1;
                state <= WAIT_TX;
                delay_timer <= 0;
            end
            WAIT_TX: begin
                tx_start <= 0;
                if (delay_timer < 16'd10000) begin
                    delay_timer <= delay_timer + 1'b1;
                end else begin
                    char_index <= char_index + 1'b1;
                    state <= GET_CHAR;
                end
            end
        endcase
    end
end
