module main(
    input clk,
    input rst_n,
    input uart_rx,
    output uart_tx,
    output [1:0] led
);

    wire [7:0] data_out;
    wire received;

    uart uart_inst (
        .clk(clk),
        .rst(~rst_n),
        .rx(uart_rx),
        .tx(uart_tx),
        .transmit(received),
        .tx_byte(data_out),
        .received(received),
        .rx_byte(data_out),
        .is_receiving(),
        .is_transmitting(),
        .recv_error()
    );

    assign led[0] = ~data_out[0]; 
    assign led[1] = ~data_out[1];

endmodule
